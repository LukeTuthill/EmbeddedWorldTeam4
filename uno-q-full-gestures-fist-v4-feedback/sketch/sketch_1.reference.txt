// Supplied animation reference; merged into sketch.ino while retaining Python RPC.
#include <Arduino_RouterBridge.h>
#include <Arduino_Modulino.h>
#include <Arduino_LED_Matrix.h>

ModulinoPixels pixels;
ModulinoBuzzer buzzer;
Arduino_LED_Matrix matrix;

int NUM_PIXELS = 8;

int ROWS = 8;
int COLS = 13;
int MATRIX_BRIGHTNESS = 255;

const uint8_t DIGITS[10][5] = {
  {0b111, 0b101, 0b101, 0b101, 0b111},  // 0
  {0b010, 0b110, 0b010, 0b010, 0b111},  // 1
  {0b111, 0b001, 0b111, 0b100, 0b111},  // 2
  {0b111, 0b001, 0b111, 0b001, 0b111},  // 3
  {0b101, 0b101, 0b111, 0b001, 0b001},  // 4
  {0b111, 0b100, 0b111, 0b001, 0b111},  // 5
  {0b111, 0b100, 0b111, 0b101, 0b111},  // 6
  {0b111, 0b001, 0b001, 0b001, 0b001},  // 7
  {0b111, 0b101, 0b111, 0b101, 0b111},  // 8
  {0b111, 0b101, 0b111, 0b001, 0b111},  // 9
};

// Used for both leds and buzzer
long duration = 1000;

int frequency = 440;

int animation_time = 0;

int loop_interval = 10;

enum Gesture {
    RAISED_PALM,
    THUMBS_UP,
    SWIPE,
    NONE
};

Gesture current_gesture = NONE;
Gesture last_gesture = NONE;
int gesture_count = 0;

void setup() {
    Bridge.begin();

    Bridge.provide_safe("raised_palm", raised_palm);
    Bridge.provide_safe("thumbs_up", thumbs_up);
    Bridge.provide_safe("swipe", swipe);

    Monitor.begin();
    Modulino.begin(Wire1);

    matrix.begin();
    matrix.setGrayscaleBits(8);

    if (!pixels.begin()) Monitor.println("Pixels not found!");
    if (!buzzer.begin()) Monitor.println("Buzzer not found!");
}

void display_count() {
    int n = gesture_count;
    if (n > 999) n = 999;

    // Split into digits (least significant first)
    uint8_t digits[3];
    uint8_t len = 0;
    do {
        digits[len++] = n % 10;
        n /= 10;
    } while (n > 0);

    const uint8_t width = len * 4 - 1;  // 3 px per digit + 1 px gap
    const uint8_t x0 = (COLS - width) / 2;
    const uint8_t y0 = (ROWS - 5) / 2;

    uint8_t frame[ROWS * COLS] = {0};  // row-major: index = row * COLS + col

    for (uint8_t i = 0; i < len; i++) {
        const uint8_t d = digits[len - 1 - i];  // most significant first
        for (uint8_t row = 0; row < 5; row++) {
            for (uint8_t col = 0; col < 3; col++) {
                if (DIGITS[d][row] & (0b100 >> col)) {
                    frame[(y0 + row) * COLS + x0 + i * 4 + col] = MATRIX_BRIGHTNESS;
                }
            }
        }
    }
    matrix.draw(frame);
}

void reset_pixels() {
    delay(1);
    pixels.clear();
    pixels.show();
}

void raised_palm_animation() {
    int phase = animation_time % 100;
    if (phase == 0) {
        for (int i = 0; i < NUM_PIXELS; i++) {
            delay(1);
            pixels.set(i, RED, 50);
        }
        delay(1);
        pixels.show();
        buzzer.tone(frequency, 50);
    }
    else if (phase == 50) {
        reset_pixels();
    }
}

void swipe_animation() {
    const int trailLength = 4;
    const int passes = 2;

    long totalDistance = (long)NUM_PIXELS * passes;
    long head = (animation_time * totalDistance) / duration;

    pixels.clear();
    for (int trail = 0; trail < trailLength; trail++) {
        long idx = head - trail;
        idx = ((idx % NUM_PIXELS) + NUM_PIXELS) % NUM_PIXELS; // wrap into [0, numPixels)
        int brightness = 100 - trail * 20;
        pixels.set((int)idx, BLUE, brightness);
    }
    pixels.show();

    int buzz_frequency = 440 / (head + 1);
    buzzer.tone(buzz_frequency, loop_interval + 2);
}



void thumbs_up_animation() {
    float timeFactor = 1.0 - ((float)animation_time / (float)duration); // 1 -> 0
    timeFactor *= timeFactor; // Squaring for smoothness
    if (timeFactor < 0) timeFactor = 0;

    int brightness = (int)(100 * timeFactor);

    for (int i = 0; i < NUM_PIXELS; i++) {
        pixels.set(i, GREEN, brightness);
    }
    
    pixels.show();
    int buzz_frequency = (int)(frequency * timeFactor);
    buzzer.tone(buzz_frequency, loop_interval + 2);
}

void update_animation() {
    if (animation_time >= duration) {
        current_gesture = NONE;
        reset_pixels();
        return;
    }

    switch(current_gesture) {
        case RAISED_PALM:
            raised_palm_animation();
            break;
        case THUMBS_UP:
            thumbs_up_animation();
            break;
        case SWIPE:
            swipe_animation();
            break;
        default:
            return;
    }
}

void start_gesture(Gesture gesture) {
    reset_pixels();

    current_gesture = gesture;
    animation_time = 0;

    if (gesture != last_gesture) {
        gesture_count++;
        last_gesture = gesture;
        display_count();
    }

    update_animation();
}

void loop() {
    if (current_gesture != NONE) {
        animation_time += 10;
        update_animation();
    }
    delay(loop_interval);
}

// API for python code to call when the given gestures are done
void raised_palm() {
    Monitor.println("Raised Palm");
    start_gesture(RAISED_PALM);
}

void thumbs_up() {
    Monitor.println("Thumbs up");
    start_gesture(THUMBS_UP);
}

void swipe() {
    Monitor.println("Swipe");
    start_gesture(SWIPE);
}
